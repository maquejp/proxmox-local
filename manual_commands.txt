mkdir -p /var/lib/vz/template/qcow2
cd /var/lib/vz/template/qcow2

wget https://dl.rockylinux.org/pub/rocky/10/images/x86_64/Rocky-10-GenericCloud-Base-10.2-20260525.0.x86_64.qcow2

wget https://dl.rockylinux.org/pub/rocky/10/images/x86_64/Rocky-10-GenericCloud-Base-10.2-20260525.0.x86_64.qcow2.CHECKSUM

sha256sum Rocky-10-GenericCloud-Base-10.2-20260525.0.x86_64.qcow2
cat Rocky-10-GenericCloud-Base-10.2-20260525.0.x86_64.qcow2.CHECKSUM

qm create 200 \
  --name rocky-test \
  --machine q35 \
  --bios seabios \
  --memory 16384 \
  --cores 6 \
  --cpu host \
  --net0 virtio,bridge=vmbr0,firewall=1

qm importdisk 200 \
  Rocky-10-GenericCloud-Base-10.2-20260525.0.x86_64.qcow2 \
  local-lvm

qm set 200 --scsihw virtio-scsi-single
qm set 200 --scsi0 local-lvm:vm-200-disk-0,iothread=1

qm resize 200 scsi0 60G

qm set 200 --ide2 local-lvm:cloudinit

qm set 200 --boot order=scsi0

qm set 200 --agent enabled=1

qm set 200 \
  --ciuser dev \
  --sshkeys /root/.ssh/id_ed25519.pub

  qm set 200 \
  --ipconfig0 ip=192.168.1.200/24,gw=192.168.1.1

  qm cloudinit update 200

  qm cloudinit dump 200 user

  qm cloudinit dump 200 network

  qm start 200
